function Orbit(o){
  const s = Object.assign({r:3, theta:0.6, phi:1.1, min:1.3, max:8, target:new THREE.Vector3(), auto:false}, o);
  let mode = null, px = 0, py = 0, pinch = 0;
  s.apply = function(){ s.phi = Math.max(0.05, Math.min(Math.PI / 2 - 0.02 + (s.globe ? Math.PI / 2 : 0), s.phi));
    cam.position.set(s.target.x + s.r * Math.sin(s.phi) * Math.sin(s.theta), s.target.y + s.r * Math.cos(s.phi), s.target.z + s.r * Math.sin(s.phi) * Math.cos(s.theta)); cam.lookAt(s.target); };
  canvas.addEventListener('contextmenu', e => e.preventDefault());
  canvas.addEventListener('pointerdown', e => { s.targetTheta = s.targetPhi = s.targetR = undefined; mode = (e.button === 2 || e.shiftKey) ? 'pan' : 'rot'; px = e.clientX; py = e.clientY; canvas.setPointerCapture(e.pointerId); s.moved = false; });
  canvas.addEventListener('pointerup', e => { mode = null; });
  canvas.addEventListener('pointermove', e => { if (!mode || !s.enabled) return; const dx = e.clientX - px, dy = e.clientY - py; px = e.clientX; py = e.clientY; if (Math.abs(dx) + Math.abs(dy) > 1) s.moved = true;
    if (mode === 'rot'){ s.theta -= dx * 0.006 * (s.globe ? s.r / 3 : 1); s.phi -= dy * 0.006 * (s.globe ? s.r / 3 : 1); }
    else { const k = s.r * 0.0016; const right = new THREE.Vector3().setFromMatrixColumn(cam.matrix, 0), fwd = new THREE.Vector3(-Math.sin(s.theta), 0, -Math.cos(s.theta));
      s.target.addScaledVector(right, -dx * k).addScaledVector(fwd, dy * k); } });
  canvas.addEventListener('wheel', e => { if (!s.enabled) return; e.preventDefault(); s.targetR = undefined; s.r = Math.max(s.min, Math.min(s.max, s.r * (1 + Math.sign(e.deltaY) * 0.1))); }, {passive:false});
  canvas.addEventListener('touchmove', e => { if (e.touches.length == 2 && s.enabled){ const d = Math.hypot(e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY); if (pinch) s.r = Math.max(s.min, Math.min(s.max, s.r * pinch / d)); pinch = d; } }, {passive:true});
  canvas.addEventListener('touchend', () => { pinch = 0; });
  s.enabled = true; return s;
}

async function loadEarthTiles(lat, lon, z){
  const N = 3, tw = 256, n = N * tw; const cx = lon2x(lon, z), cy = lat2y(lat, z); const tx0 = Math.floor(cx) - 1, ty0 = Math.floor(cy) - 1;
  const elev = new Float32Array(n * n); const eg = document.createElement('canvas'); eg.width = n; eg.height = n; const ec = eg.getContext('2d', {willReadFrequently:true});
  const ic = document.createElement('canvas'); ic.width = n; ic.height = n; const ig = ic.getContext('2d'); ig.fillStyle = '#263B59'; ig.fillRect(0, 0, n, n);
  let eOk = 0, iOk = 0; const jobs = [];
  for (let j = 0; j < N; j++) for (let i = 0; i < N; i++){
    const o = {z:z, x:tx0 + i, y:ty0 + j};
    jobs.push(loadImage(fill(D.tiles.terrain, o)).then(im => { ec.drawImage(im, i * tw, j * tw); eOk++; }).catch(() => {}));
    if (D.tiles.imagery) jobs.push(loadImage(fill(D.tiles.imagery, o)).then(im => { ig.drawImage(im, i * tw, j * tw, tw, tw); iOk++; }).catch(() => {}));
  }
  status('Loading terrain tiles (z' + z + ') ...'); await Promise.all(jobs);
  if (eOk < N * N) return {error:'Elevation tiles unavailable (' + eOk + '/' + (N * N) + ' loaded). Check the network connection; no terrain is invented.'};
  let px; try { px = ec.getImageData(0, 0, n, n).data; } catch (e) { return {error:'Elevation tiles blocked by the browser (CORS).'}; }
  for (let k = 0; k < n * n; k++) elev[k] = px[k * 4] * 256 + px[k * 4 + 1] + px[k * 4 + 2] / 256 - 32768;
  const mpp = 156543.03392 * Math.cos(lat * RAD) / Math.pow(2, z);
  return {n:n, elev:blur(elev, n), mpp:mpp, unit:(n * mpp) / 10, tx0:tx0, ty0:ty0, N:N, z:z, img:(iOk == N * N ? ic : null), imgCount:iOk, source:'earth'};
}
function gridTerrain(g){ const n = g.n; return {n:n, elev:Float32Array.from(g.h), mpp:g.cell_m, unit:(n * g.cell_m) / 10, N:1, img:null, source:'grid', schematic:g.schematic, label:g.label}; }

function stats(e){ let lo = 1e9, hi = -1e9; for (let k = 0; k < e.length; k++){ if (e[k] < lo) lo = e[k]; if (e[k] > hi) hi = e[k]; } return [lo, hi]; }

function colorFor(layerName, i, T, lo, hi, slopeArr){
  const e = T.elev[i];
  if (layerName === 'elevation'){
    if (e <= 0 && T.source === 'earth'){ const t = Math.min(1, -e / 200); return R.water[3].clone().lerp(R.water[0], t); }
    const t = (e - Math.max(lo, 0)) / Math.max(hi - Math.max(lo, 0), 1); // land: vegetation -> land -> ice
    if (T.source === 'earth') return ramp([R.veg[1], R.veg[2], R.veg[3], R.land[2], R.land[3], R.land[4], R.ice[0]], t);
    return ramp([R.land[0], R.land[1], R.land[2], R.land[3], R.land[4], R.ice[0]], (e - lo) / Math.max(hi - lo, 1));
  }
  if (layerName === 'slope') return ramp(R.slope, slopeArr[i] / 45);
  if (layerName === 'volcanic') return ramp(R.volcanic, (e - lo) / Math.max(hi - lo, 1));
  return hex('#C9D3E3'); // hillshade base
}

function buildTerrain(t){
  if (T && T.mesh){ terrain.remove(T.mesh); T.mesh.geometry.dispose(); }
  T = t; const n = t.n, step = Math.max(2, Math.floor(n / 220)), seg = Math.floor((n - 1) / step);
  const [lo, hi] = stats(t.elev); t.lo = lo; t.hi = hi;
  const slope = new Float32Array(n * n); for (let y = 1; y < n - 1; y++) for (let x = 1; x < n - 1; x++){ const gx = (t.elev[y * n + x + 1] - t.elev[y * n + x - 1]) / (2 * t.mpp), gy = (t.elev[(y + 1) * n + x] - t.elev[(y - 1) * n + x]) / (2 * t.mpp); slope[y * n + x] = Math.atan(Math.hypot(gx, gy)) / RAD; }
  t.slope = slope; t.step = step; t.seg = seg;
  const geo = new THREE.BufferGeometry(); const nv = (seg + 1) * (seg + 1); const pos = new Float32Array(nv * 3), uv = new Float32Array(nv * 2), idx = [];
  t.idxMap = new Int32Array(nv);
  for (let j = 0; j <= seg; j++) for (let i = 0; i <= seg; i++){ const v = j * (seg + 1) + i; const px = Math.min(i * step, n - 1), py = Math.min(j * step, n - 1); t.idxMap[v] = py * n + px;
    pos[v * 3] = (i / seg - 0.5) * 10; pos[v * 3 + 2] = (j / seg - 0.5) * 10; uv[v * 2] = i / seg; uv[v * 2 + 1] = 1 - j / seg; }
  for (let j = 0; j < seg; j++) for (let i = 0; i < seg; i++){ const a = j * (seg + 1) + i, b = a + 1, c = a + seg + 1, d = c + 1; idx.push(a, c, b, b, c, d); }
  geo.setAttribute('position', new THREE.BufferAttribute(pos, 3)); geo.setAttribute('uv', new THREE.BufferAttribute(uv, 2)); geo.setAttribute('color', new THREE.BufferAttribute(new Float32Array(nv * 3), 3)); geo.setIndex(idx);
  t.geo = geo; t.matSat = new THREE.MeshStandardMaterial({map: t.img ? new THREE.CanvasTexture(t.img) : null, roughness:0.9, metalness:0, side:THREE.DoubleSide});
  if (t.matSat.map) t.matSat.map.anisotropy = 8;
  t.matCol = new THREE.MeshStandardMaterial({vertexColors:true, roughness:0.88, metalness:0.02, side:THREE.DoubleSide});
  t.mesh = new THREE.Mesh(geo, t.matCol); terrain.add(t.mesh);
  applyLayer(); applyExag(); drawTerrainMarkers();
  const [a, b] = [lo.toFixed(0), hi.toFixed(0)];
  $('tinfo').innerHTML = (t.source === 'earth' ? 'z' + t.z + ' - ' + (t.mpp).toFixed(0) + ' m/px - area ' + (10 * t.unit / 1000).toFixed(1) + ' km' : (t.schematic ? 'SCHEMATIC terrain - not measured' : 'user DEM') + ' - cell ' + t.mpp.toFixed(1) + ' m') + '<br>elevation ' + a + ' to ' + b + ' m';
  terrCtl.r = 11; terrCtl.theta = 0.6; terrCtl.phi = 1.0; terrCtl.target.set(0, 0, 0); legend();
}

async function setView(v, lat, lon){
  if (v === 'terrain'){
    if (D.body === 'Earth'){ if (lat !== undefined){ curLat = lat; curLon = lon; } view = 'terrain'; syncView(); const r = await loadEarthTiles(curLat, curLon, curZ); if (r.error){ status(r.error, 'err'); view = 'globe'; syncView(); return; } buildTerrain(r);
      status('Terrain: AWS Terrain Tiles (SRTM/3DEP blend) ' + (r.img ? '+ Esri World Imagery' : '(imagery unavailable; showing elevation colours)'), r.img ? '' : 'warn'); if (!r.img) { layer = 'elevation'; $('layer').value = 'elevation'; applyLayer(); } }
    else if (D.grid){ view = 'terrain'; syncView(); buildTerrain(gridTerrain(D.grid)); layer = 'elevation'; $('layer').value = 'elevation'; applyLayer(); status(D.grid.schematic ? 'Schematic relief (no measured DEM loaded for this body). Upload a DEM in Data Sources to replace it.' : 'Uploaded DEM: ' + (D.grid.label || ''), D.grid.schematic ? 'warn' : ''); }
    else { status('No local DEM for ' + D.body + '. Upload a LOLA/MOLA/HiRISE DEM in the Data Sources panel to enable terrain view.', 'warn'); return; }
  } else { view = 'globe'; syncView(); }
}
